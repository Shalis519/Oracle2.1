// lib/oracle.ts

export async function computeDailyForecast(
  birthDate: Date,
  natalChart: any,
  transits: TransitAspect[],
  date: Date
): Promise<DailyForecast> {
  
  // ===== ТОЛЬКО ОНТОЛОГИЯ =====
  
  // 1. Загружаем онтологию
  await semanticEngine.loadAll();
  
  // 2. Берем сильные транзиты
  const topTransits = transits
    .filter(t => Math.abs(t.orb) < 3)
    .sort((a, b) => Math.abs(a.orb) - Math.abs(b.orb))
    .slice(0, 3);
  
  if (topTransits.length === 0) {
    return {
      synthesisText: 'Сегодня нет сильных аспектов для прогноза.',
      transits: [],
      arcana: getArcanaOfDay(date),
      conflicts: [],
      warnings: [],
      generatedBy: 'ontology'
    };
  }
  
  // 3. Генерируем прогноз для каждого транзита
  const forecasts: string[] = [];
  const failedTransits: string[] = [];
  
  for (const transit of topTransits) {
    const context = {
      planet: transit.planet,
      sign: transit.sign,
      house: transit.house,
      aspect: transit.aspect,
      aspectPlanet: transit.aspectPlanet,
      orb: transit.orb,
      currentDate: date
    };
    
    const forecast = futuristicGenerator.generate(context);
    
    if (forecast) {
      forecasts.push(forecast);
    } else {
      // Запоминаем, для чего не нашлось данных
      const key = `${transit.planet} ${transit.aspect} ${transit.aspectPlanet || ''}`.trim();
      failedTransits.push(key);
    }
  }
  
  // 4. Формируем результат
  if (forecasts.length === 0) {
    // НЕТ ДАННЫХ В ОНТОЛОГИИ
    const missingData = failedTransits.join(', ');
    return {
      synthesisText: `Для транзитов (${missingData}) пока нет семантических данных в Oracle Studio. 
                      Администратор заполняет онтологию. Прогнозы появятся после добавления данных.`,
      transits: topTransits.map(t => ({
        ...t,
        description: 'нет данных в онтологии'
      })),
      arcana: getArcanaOfDay(date),
      conflicts: [],
      warnings: ['Онтология заполнена не полностью'],
      generatedBy: 'ontology_partial'
    };
  }
  
  // 5. Есть хотя бы один прогноз из онтологии
  let synthesisText = forecasts.join('\n\n---\n\n');
  
  if (failedTransits.length > 0) {
    synthesisText += `\n\n⚠️ Для транзитов (${failedTransits.join(', ')}) пока нет данных в онтологии.`;
  }
  
  return {
    synthesisText,
    transits: topTransits.map(t => ({
      ...t,
      description: t.description || 'данные в онтологии'
    })),
    arcana: getArcanaOfDay(date),
    conflicts: [],
    warnings: failedTransits.length > 0 ? ['Онтология заполнена не полностью'] : [],
    generatedBy: 'ontology'
  };
}
